import{epubReader}from'<APP_ROOT>/src/lib/reading/reader/epub.ts';
import{physicalPdfPages}from'<APP_ROOT>/src/lib/reading/reader/physical-pdf.ts';
const host=document.querySelector('#host') as HTMLElement,state=document.querySelector('#state')!;
const create=URL.createObjectURL,revoke=URL.revokeObjectURL,active=new Set<string>();let made=0,revoked=0;

let last:any,owner:any,abort:AbortController|undefined;

const options={host,source:format==='epub'?'/war-and-peace.epub':'/postgresql-17.pdf',format,location:resume?last:null,signal:abort.signal,changed:(x:any)=>{changes.push(x);last=x;},progress:(x:any)=>progress.push(x)};

owner=await physicalPdfPages(options as any);const total=owner.location(0).total,rendered=[];
for(const index of[0,Math.floor(total/2),total-1]){const canvas=await owner.page(index,abort.signal),ctx=canvas.getContext('2d')!,pixels=ctx.getImageData(0,0,canvas.width,canvas.height).data;let dark=0;for(let n=0;n<pixels.length;n+=4)if(pixels[n]<200&&pixels[n+1]<200&&pixels[n+2]<200)dark++;rendered.push({index,width:canvas.width,height:canvas.height,darkPixels:dark,location:owner.location(index)});host.replaceChildren(canvas);}
state.textContent='PDF rendered '+total+' pages';return{format,ms:performance.now()-at,total,boundFaces:owner.count,rendered,urls:{made,revoked,active:active.size}};
},async close(){if(owner){owner.destroy?.();owner.close?.();owner=null;}abort?.abort();abort=undefined;host.replaceChildren();await new Promise(r=>setTimeout(r,200));return{made,revoked,active:active.size};},async cancelled(format:string){const a=new AbortController();a.abort();try{await (format==='epub'?epubReader:physicalPdfPages)({host,source:format==='epub'?'/war-and-peace.epub':'/postgresql-17.pdf',format,location:null,signal:a.signal,changed(){}}as any);return{denied:false};}catch(e:any){return{denied:e.name==='AbortError',name:e.name,active:active.size};}}};
